/// Lime number circle + text. "Prune: cut..." shows "Prune:" in bold.
class _NumberedItem extends StatelessWidget {
  final String number;
  final String text;
  final bool showLine;

  const _NumberedItem({
    required this.number,
    required this.text,
    this.showLine = false,
  });

  @override
  Widget build(BuildContext context) {
    // Bold the short word before ":" (e.g. "Prune:")
    final i = text.indexOf(':');
    final hasLead = i > 0 && i < 25;
    final lead = hasLead ? text.substring(0, i + 1) : '';
    final rest = hasLead ? text.substring(i + 1) : text;

    return IntrinsicHeight(
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          SizedBox(
            width: 28,
            child: Column(
              children: [
                Container(
                  width: 28,
                  height: 28,
                  alignment: Alignment.center,
                  decoration: const BoxDecoration(
                    color: AppColors.lime,
                    shape: BoxShape.circle,
                  ),
                  child: Text(
                    number,
                    style: const TextStyle(
                      color: AppColors.primary,
                      fontSize: 13,
                      fontWeight: FontWeight.w800,
                    ),
                  ),
                ),
                if (showLine)
                  Expanded(
                    child: Container(
                      width: 2,
                      margin: const EdgeInsets.symmetric(vertical: 3),
                      color: AppColors.inputBorder,
                    ),
                  ),
              ],
            ),
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Padding(
              padding: EdgeInsets.only(top: 4, bottom: showLine ? 14 : 0),
              child: Text.rich(
                TextSpan(
                  style: const TextStyle(
                    color: AppColors.mutedText,
                    fontSize: 15,
                    height: 1.4,
                  ),
                  children: [
                    if (hasLead)
                      TextSpan(
                        text: lead,
                        style: const TextStyle(
                          color: AppColors.textDark,
                          fontWeight: FontWeight.w700,
                        ),
                      ),
                    TextSpan(text: rest),
                  ],
                ),
              ),
            ),
          ),
        ],
      ),
    );
  }
}

enum _PillStyle { dark, outline, soft }

/// Full-width pill button in three styles.
class _PillButton extends StatelessWidget {
  const _PillButton({
    required this.label,
    required this.icon,
    required this.style,
    required this.onPressed,
  });

  final String label;
  final IconData icon;
  final _PillStyle style;
  final VoidCallback onPressed;

  @override
  Widget build(BuildContext context) {
    final Color bg = switch (style) {
      _PillStyle.dark => AppColors.primary,
      _PillStyle.outline => Colors.white,
      _PillStyle.soft => const Color(0xFFE6F0E6),
    };
    final Color fg = style == _PillStyle.dark ? Colors.white : AppColors.primary;
    final Color iconColor =
        style == _PillStyle.dark ? AppColors.lime : AppColors.primary;

    return SizedBox(
      width: double.infinity,
      height: 58,
      child: OutlinedButton.icon(
        onPressed: onPressed,
        icon: Icon(icon, color: iconColor, size: 22),
        label: Text(
          label,
          style: TextStyle(
            color: fg,
            fontSize: 17,
            fontWeight: FontWeight.w700,
          ),
        ),
        style: OutlinedButton.styleFrom(
          backgroundColor: bg,
          shape: const StadiumBorder(),
          side: style == _PillStyle.outline
              ? const BorderSide(color: AppColors.primary, width: 1.6)
              : BorderSide.none,
        ),
      ),
    );
  }
}
